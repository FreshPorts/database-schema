CREATE OR REPLACE FUNCTION check_siblings() returns OPAQUE as '
   declare 
      ElementCount integer;
		PortCount		integer;
		OldParentStatus	text;
begin

  if new.Parent_ID is null then
    select count(*) 
      into ElementCount
      from element
     where element.name=new.name
       and element.Parent_ID is null
     group by element.name;
  else
    select count(*) 
      into ElementCount
      from element
     where element.name=new.name
       and element.Parent_ID=new.Parent_ID
     group by element.name;
  end if;

  if (ElementCount > 1) then
    RAISE EXCEPTION ''element % cannot be created as it already exists under that parent %'', new.name, new.parent_id;
  end if;

	-- are we dealing with a Makefile which has a parent?

	if new.name = ''Makefile'' and new.Parent_ID is NOT NULL then

		-- is this Makefile part of a port?
		SELECT count(*)
		  INTO PortCount
		  FROM ports
		 WHERE element_id = new.Parent_ID;

		IF PortCount > 0 THEN


		-- we did a truth table.  The status of a port is
		-- determined by the status of the Makefile.
		-- if it is deleted, we delete the port.
		-- and if the Makefile is put back, we activate the port

			SELECT status
			  INTO oldParentStatus
			  FROM element
			 WHERE id = new.Parent_ID;

			if oldParentStatus != new.status then
				UPDATE element
				   SET status = new.status
				 WHERE id = new.Parent_ID;
			end if;
		end if;
	end if;

  RETURN NEW;

end;
' LANGUAGE 'plpgsql';


  drop trigger check_siblings on element;
create trigger check_siblings after insert or update 
on element
for each row
execute procedure check_siblings();

end;

  drop trigger on_delete_remove_children on element;

on element for each row
execute procedure on_delete_remove_children();



CREATE OR REPLACE FUNCTION check_parent() returns OPAQUE as '
   declare
      ElementCount int4;
begin
   --  message ''check_parent'';
   if new.parent_id is not null then
      --    message ''check_parent 2'';
      select count(*) 
        into ElementCount
        from element
       where element.id = new.parent_id;
      --    message ''check_parent 3'';
      if (ElementCount = 0) then
         -- message ''no such parent '' || NOW(*);
         RAISE EXCEPTION ''modifications to element % with name=% cannot be completed as the parent % does not exist.'', new.id, new.name, new.parent_id;
      end if;
      if new.parent_id = new.id then
         RAISE EXCEPTION ''modifications to element % with name=% cannot be completed as the an element cannot be a child of itself.'', new.id, new.name, new.parent_id;
      end if;
   end if;
   --    message ''check_parent 4''

   RETURN NEW;
end;
' LANGUAGE 'plpgsql';

  drop trigger check_parent on element;
create trigger check_parent before insert or update
on element for each row
execute procedure check_parent();



CREATE OR REPLACE FUNCTION parent_must_be_directory() returns OPAQUE as '
   declare
      ParentType char(1);
begin
   --  message ''parent_must_be_directory'';

   if new.Parent_Id is not null then
      select directory_file_flag 
        into ParentType
       from element
      where element.id = new.parent_id;

      if (ParentType = ''F'') then
         -- message ''parent cannot be a file '' || NOW(*);
         RAISE NOTICE ''modifications to element % with name=% required modifying the the parent % from a file to directory.'', new.id, new.name, new.parent_id;
		UPDATE element
		   SET directory_file_flag = ''D''
		 WHERE element.id          = new.parent_id;
     end if;
   end if;

   RETURN NEW;
end;
' LANGUAGE 'plpgsql';

drop trigger parent_must_be_directory on element;
create trigger parent_must_be_directory before insert or update
on element for each row
execute procedure parent_must_be_directory();

CREATE OR REPLACE FUNCTION id_change() returns OPAQUE as '
begin
   --  message ''check_siblings 4'';
   if (new.id <> old.id) then
      -- message ''change of id not allowed '' || NOW(*);
      RAISE EXCEPTION ''modifications to element % with name=% cannot be completed as you are not allowed to change id (new value was %).'', old.id, old.name, new.id;
   end if;

   RETURN NEW;
end;
' LANGUAGE 'plpgsql';

-- the trigger
  drop trigger id_change on element;
create trigger id_change before update
on element for each row
execute procedure id_change();


CREATE OR REPLACE FUNCTION version_revision_uniqueness() returns OPAQUE as '
   declare 
      ElementID            int4;
      ExistingRevisionName text;
begin

   select revision_name
     into ExistingRevisionName
     from system_branch_element_revision
    where system_branch_id = new.system_branch_id
      and element_id        = new.element_id;

   IF FOUND THEN
      delete from system_branch_element_revision 
       where system_branch_id = new.system_branch_id
         and element_id       = new.element_id;
   END IF;

   RETURN NEW;

end;
' LANGUAGE 'plpgsql';



  drop trigger version_revision_uniqueness on system_branch_element_revision;
create trigger version_revision_uniqueness before insert or update 
on system_branch_element_revision
for each row
execute procedure version_revision_uniqueness();

CREATE OR REPLACE FUNCTION element_delete_check() returns OPAQUE as '
begin

   -- if we are removing an element, make sure we set the
   -- element status to deleted.

   if (NEW.change_type = \'R\') then
--      select Element_SetStatus(new.element_id, \'D\');
      update element
         set status = \'D\'
       where id     = NEW.element_id;
   end if;

   --
   -- * * * * * * * * * * * * * * * *
   -- use a different return value if you change this
   -- function to a BEFORE
   -- * * * * * * * * * * * * * * * *
   RETURN NEW;
end;
' LANGUAGE 'plpgsql';


-- the trigger
  drop trigger element_delete_check on commit_log_elements;
create trigger element_delete_check before insert or update
on commit_log_elements for each row
execute procedure element_delete_check();

create or replace function element_update_set_children(int4, text) returns int as 
'DECLARE

   node ALIAS for $1;
   path ALIAS for $2;
   children record;

   BEGIN
--      RAISE NOTICE ''PROCESSING all children of % ...'', path;
      FOR children IN SELECT ep.id, ep.pathname, e.name, e.directory_file_flag
                        FROM element_pathnames ep, element e
                       WHERE ep.id       = e.id
                         AND e.parent_id = node LOOP
--         RAISE NOTICE ''   updating %'', children.name ;
         UPDATE element_pathnames set pathname = path ||  ''/'' || children.name where id = children.id;

         -- no sense doing recursive calls on files, just directories
         IF children.directory_file_flag = ''D'' THEN
            -- recursive call
            perform element_update_set_children(children.id, path ||  ''/'' || children.name);
         END IF;
      END LOOP;

      return 0;
   END;'

language 'plpgsql';

create or replace function element_update() returns opaque as '
DECLARE
   parent_pathname   text;
   my_pathname       text;
   RowsModified      int4;
BEGIN
--   RAISE NOTICE ''TRIGGER: in element_update % %'', old.name, new.name;
   IF old.name <> new.name THEN
      select pathname
        into parent_pathname
        from element_pathnames
       where id = new.parent_id;
      IF found THEN
         my_pathname =  parent_pathname || \'/\' ||new.name;
      ELSE
         my_pathname = \'/\' || new.name;
      END IF;
      
      UPDATE element_pathnames SET pathname = my_pathname WHERE id = new.id;
      GET DIAGNOSTICS RowsModified = ROW_COUNT;
      IF RowsModified <> 1 THEN
         RAISE EXCEPTION ''Row in element_pathnames not found for %'', new.id;
      END IF;

      perform element_update_set_children(new.id, my_pathname);
   END IF;

   RETURN new;
END;'
language 'plpgsql';


  drop trigger element_update on element;
create trigger element_update before update on element for each row
execute procedure element_update();




create or replace function element_insert() returns opaque as '
   DECLARE
      parent_pathname   text;
      my_pathname       text;
   BEGIN
      select pathname
        into parent_pathname
        from element_pathnames
       where id = new.parent_id;


      if found then
         my_pathname =  parent_pathname || \'/\' ||new.name;
      else
         my_pathname = \'/\' || new.name;
      end if;

      INSERT into element_pathnames (id, pathname) values (new.id, my_pathname);

      RETURN new;
   END;'

language 'plpgsql';


  drop trigger element_insert on element;
create trigger element_insert after insert on element for each row
execute procedure element_insert();
